#+begin_src powershell
# Single vs double quotes
$name = 'Trevor'
"Hello $name"                                             # Double quotes interpolate variables → "Hello Trevor"

'HELLO'.ToLower()                                         # → "hello"
'hello world'.IndexOf('world')                            # Returns 6
'hello world'.Substring(6)                                # Returns 'world'
'hello world'.Contains('world')                           # Returns $true

#+begin_src powershell
# Basic function

# Advanced function — the production-grade pattern
function Get-ServerStatus {
    [CmdletBinding()]
    param (
        [Parameter(Mandatory, ValueFromPipeline, ValueFromPipelineByPropertyName)]
        [string]$ComputerName,

        [Parameter()]
        [int]$Port = 443,

        [Parameter()]
        [ValidateSet('HTTP','HTTPS','SSH')]
        [string]$Protocol = 'HTTPS'
    )
    begin {
        # Runs once before pipeline input — use for setup/init
    }
    process {
        # Runs once per pipeline object — $ComputerName is the current item
        Write-Verbose "Checking $ComputerName on $Protocol:$Port"
        Test-NetConnection -ComputerName $ComputerName -Port $Port
    }
    end {
        # Runs once after all pipeline input — use for cleanup/summary
    }
}

# Splatting — pass a hashtable as parameters (cleaner than long lines)
$Params = @{
    ComputerName = 'server01'
    Port         = 22
    Protocol     = 'SSH'
}
Get-ServerStatus @Params

* Error Handling
:PROPERTIES:
:DESCRIPTION: try/catch/finally, error variables, terminating vs non-terminating errors
:END:

#+begin_src powershell
# try / catch / finally
try {
    Get-Item 'C:\nonexistent' -ErrorAction Stop           # -ErrorAction Stop makes non-terminating errors catchable
} catch [System.IO.FileNotFoundException] {
    Write-Warning "File not found: $_"
} catch {
    Write-Error "Unexpected error: $($_.Exception.Message)"
} finally {
    # Always runs — use for cleanup (close connections, etc.)
    Write-Verbose 'Cleanup complete'
}

# Error action preferences
Get-Item 'missing' -ErrorAction SilentlyContinue          # Suppress error, continue execution
Get-Item 'missing' -ErrorAction Stop                      # Promote to terminating error (catchable)
Get-Item 'missing' -ErrorAction Ignore                    # Suppress error AND don't add to $Error

# Capture error in a variable
Get-Item 'missing' -ErrorVariable myErr -ErrorAction SilentlyContinue
$myErr[0].Exception.Message                               # Inspect the captured error

# Inspect last error
$Error[0]                                                 # Most recent error object
$Error[0].Exception.Message                               # Just the message string
$Error[0].InvocationInfo.Line                             # The line that caused the error
Get-Error                                                 # [PS7+] Detailed, formatted error view for $Error[0]

# Terminating vs non-terminating
# Non-terminating: command reports an error but execution continues (default for most cmdlets)
# Terminating: execution stops immediately unless caught in try/catch
# Use -ErrorAction Stop to convert non-terminating → terminating
#+end_src

* Classes
:PROPERTIES:
:DESCRIPTION: Class definition, constructors, methods, inheritance
:END:

#+begin_src powershell
class NetworkDevice {
    [string]   $Hostname
    [string]   $IPAddress
    [string]   $Role      = 'Unknown'
    [bool]     $Reachable = $false

    # Default constructor
    NetworkDevice() {}

    # Parameterized constructor
    NetworkDevice([string]$Hostname, [string]$IPAddress) {
        $this.Hostname  = $Hostname
        $this.IPAddress = $IPAddress
    }

    # Method
    [bool] Ping() {
        $result = Test-Connection -ComputerName $this.IPAddress -Count 1 -Quiet
        $this.Reachable = $result
        return $result
    }

    [string] ToString() {
        return '{0} ({1}) — Reachable: {2}' -f $this.Hostname, $this.IPAddress, $this.Reachable
    }
}

$dev = [NetworkDevice]::new('core-sw-01', '10.0.0.254')
$dev.Role = 'Core Switch'
$dev.Ping()
$dev.ToString()

# Inheritance
class ManagedSwitch : NetworkDevice {
    [int] $PortCount

    ManagedSwitch([string]$Hostname, [string]$IPAddress, [int]$PortCount)
        : base($Hostname, $IPAddress) {
        $this.PortCount = $PortCount
    }
}

* REST APIs

** Invoke-RestMethod vs Invoke-WebRequest
#+begin_src powershell
# Invoke-RestMethod   — parses JSON/XML automatically, returns objects. Use this for APIs.
# Invoke-WebRequest   — returns raw HTTP response (StatusCode, Headers, Content as string).
#                       Use when you need headers, status codes, or non-JSON responses.

# Basic GET
Invoke-RestMethod -Uri 'https://api.github.com/events' -Method Get

# Inspect raw response (status code, headers)
$response = Invoke-WebRequest -Uri 'https://api.github.com'
$response.StatusCode
$response.Headers['X-RateLimit-Remaining']
$response.Content                                         # Raw JSON string
#+end_src

** Authentication Patterns
#+begin_src powershell
# Bearer token (OAuth2, JWT — most modern APIs)
$headers = @{ Authorization = "Bearer $token" }
Invoke-RestMethod -Uri 'https://api.example.com/data' -Headers $headers

# API key in header
$headers = @{ 'X-API-Key' = $apiKey }
Invoke-RestMethod -Uri 'https://api.example.com/data' -Headers $headers

# Basic auth (encode credentials to Base64)
$cred    = 'user:password'
$encoded = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes($cred))
$headers = @{ Authorization = "Basic $encoded" }
Invoke-RestMethod -Uri 'https://api.example.com' -Headers $headers

# POST with JSON body
$body = @{ name = 'web01'; role = 'web' } | ConvertTo-Json
Invoke-RestMethod -Uri 'https://api.example.com/servers' `
    -Method Post `
    -Headers $headers `
    -Body $body `
    -ContentType 'application/json'

# Ignore SSL certificate errors (lab / self-signed certs only — never in prod)
Invoke-RestMethod -Uri 'https://lab-host' -SkipCertificateCheck   # [PS7+]
#+end_src

** vCenter REST API
#+begin_src powershell
# Authenticate — get session token
$vcenter = 'vcenter.lab.local'
$cred    = 'administrator@vsphere.local:VMware1!'
$encoded = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes($cred))
$authHeaders = @{ Authorization = "Basic $encoded" }

$session = Invoke-RestMethod -Uri "https://$vcenter/api/session" `
    -Method Post `
    -Headers $authHeaders `
    -SkipCertificateCheck

$apiHeaders = @{ 'vmware-api-session-id' = $session }

# List all VMs
$vms = Invoke-RestMethod -Uri "https://$vcenter/api/vcenter/vm" `
    -Headers $apiHeaders `
    -SkipCertificateCheck
$vms | Select-Object name, power_state, memory_size_MiB

# Get VM details
$vmId = $vms[0].vm
Invoke-RestMethod -Uri "https://$vcenter/api/vcenter/vm/$vmId" `
    -Headers $apiHeaders `
    -SkipCertificateCheck

# Power operations
Invoke-RestMethod -Uri "https://$vcenter/api/vcenter/vm/$vmId/power?action=start" `
    -Method Post `
    -Headers $apiHeaders `
    -SkipCertificateCheck

# List hosts
Invoke-RestMethod -Uri "https://$vcenter/api/vcenter/host" `
    -Headers $apiHeaders `
    -SkipCertificateCheck

# Terminate session
Invoke-RestMethod -Uri "https://$vcenter/api/session" `
